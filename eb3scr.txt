local function isKeyDown()
    if Settings.AimMouseButton then return UserInputService:IsMouseButtonPressed(Settings.AimMouseButton) end
    if Settings.AimKey then return UserInputService:IsKeyDown(Settings.AimKey) end
    return false
end

local function getTriggerTarget()
    local ray = Camera:ViewportPointToRay(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)
    local params = RaycastParams.new()
    params.FilterDescendantsInstances = {LocalPlayer.Character, Camera}
    params.FilterType = Enum.RaycastFilterType.Blacklist
    local res = workspace:Raycast(ray.Origin, ray.Direction * 1000, params)
    if res and res.Instance then
        local m = res.Instance:FindFirstAncestorOfClass("Model")
        local p = Players:GetPlayerFromCharacter(m)
        if p and p ~= LocalPlayer and m:FindFirstChild("Humanoid") and m.Humanoid.Health > 0 then
            if not Settings.TeamCheck or p.Team ~= LocalPlayer.Team then return true end
        end
    end
    return false
end

local function createESP(p)
    if p == LocalPlayer then return end
    local g = Instance.new("ScreenGui", CoreGui); g.IgnoreGuiInset = true
    local boxes = {}; for i=1,8 do boxes[i] = Instance.new("Frame", g); boxes[i].BorderSizePixel = 0 end
    local name = Instance.new("TextLabel", g); name.BackgroundTransparency = 1; name.Font = Enum.Font.SourceSansBold; name.TextSize = 14; applyOutline(name)
    local tHUD = Instance.new("Frame", g); tHUD.Size = UDim2.new(0,30,0,30); tHUD.BackgroundTransparency = 1; tHUD.AnchorPoint = Vector2.new(0.5,0.5)
    local tL = {}; for i=1,4 do tL[i] = Instance.new("Frame", tHUD); tL[i].BorderSizePixel = 0 end

    RunService.RenderStepped:Connect(function()
        local c = p.Character; local hrp = c and c:FindFirstChild("HumanoidRootPart")
        if hrp and c:FindFirstChild("Humanoid") and c.Humanoid.Health > 0 and (not Settings.TeamCheck or p.Team ~= LocalPlayer.Team) then
            local pos, onS = Camera:WorldToViewportPoint(hrp.Position)
            if onS then
                g.Enabled = true
                local head = c:FindFirstChild("Head")
                if not head then return end
                local h = math.abs(Camera:WorldToViewportPoint(head.Position + Vector3.new(0,0.8,0)).Y - Camera:WorldToViewportPoint(hrp.Position - Vector3.new(0,3,0)).Y)
                local w = h * 0.6; local x, y, l = pos.X - w/2, pos.Y - h/2, w * 0.25
                for i=1,8 do boxes[i].BackgroundColor3 = Settings.MainColor end
                boxes[1].Position = UDim2.new(0,x,0,y); boxes[1].Size = UDim2.new(0,l,0,2)
                boxes[2].Position = UDim2.new(0,x,0,y); boxes[2].Size = UDim2.new(0,2,0,l)
                boxes[3].Position = UDim2.new(0,x+w-l,0,y); boxes[3].Size = UDim2.new(0,l,0,2)
                boxes[4].Position = UDim2.new(0,x+w-2,0,y); boxes[4].Size = UDim2.new(0,2,0,l)
                boxes[5].Position = UDim2.new(0,x,0,y+h); boxes[5].Size = UDim2.new(0,l,0,2)
                boxes[6].Position = UDim2.new(0,x,0,y+h-l); boxes[6].Size = UDim2.new(0,2,0,l)
                boxes[7].Position = UDim2.new(0,x+w-l,0,y+h); boxes[7].Size = UDim2.new(0,l,0,2)
                boxes[8].Position = UDim2.new(0,x+w-2,0,y+h-l); boxes[8].Size = UDim2.new(0,2,0,l)
                name.Visible = true; name.Text = p.Name; name.TextColor3 = Settings.MainColor; name.Position = UDim2.new(0,x+w/2-100,0,y-20); name.Size = UDim2.new(0,200,0,20)
                
                if getgenv().curT == p and isKeyDown() then
                    tHUD.Visible = true; tHUD.Position = UDim2.new(0,pos.X,0,pos.Y); tHUD.Rotation = (tick()*400)%360

    end)

RunService.RenderStepped:Connect(function()
    FOVCircle.Position = UserInputService:GetMouseLocation()
    FOVCircle.Radius = Settings.AimbotFOV
    local target = nil; local maxD = math.huge; local mP = UserInputService:GetMouseLocation()
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild("Head") and p.Character:FindFirstChild("Humanoid") and p.Character.Humanoid.Health > 0 then
            if not Settings.TeamCheck or p.Team ~= LocalPlayer.Team then
                local sP, onS = Camera:WorldToViewportPoint(p.Character.Head.Position)
                if onS then
                    local d = (Vector2.new(sP.X, sP.Y) - mP).Magnitude
                    if d <= Settings.AimbotFOV and d < maxD then maxD = d; target = p end
                end
            end
        end
    end
    if target and Settings.AimbotEnabled and isKeyDown() then
        getgenv().curT = target
        Camera.CFrame = CFrame.new(Camera.CFrame.Position, target.Character.Head.Position)
    else getgenv().curT = nil end
    if Settings.Autofire then if getTriggerTarget() then mouse1press() else mouse1release() end end
end)
